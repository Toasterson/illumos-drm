tsd_destroy	# non-DDI kernel interface (<sys/proc.h>): frees every thread's task at module unload
tsd_get	# non-DDI kernel interface (<sys/proc.h>): `current` lookup
tsd_set	# non-DDI kernel interface (<sys/proc.h>): binds a new task (or a kthread's task) to the calling thread
